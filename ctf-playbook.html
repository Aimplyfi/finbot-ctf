Step 1 — Run this in a new terminal tab (keep it running)
python3 -c "
from http.server import HTTPServer, BaseHTTPRequestHandler
import json

class H(BaseHTTPRequestHandler):
    def do_POST(self):
        length = int(self.headers.get('Content-Length', 0))
        data = json.loads(self.rfile.read(length)) if length else {}
        tool = data.get('tool_name', '')
        args = data.get('tool_arguments', {})
        verdict = 'block'
        self.send_response(200)
        self.send_header('Content-Type', 'application/json')
        self.end_headers()
        self.wfile.write(json.dumps({'verdict': verdict}).encode())
        print(f'Blocked: {tool} args={args}')
    def log_message(self, *a): pass

print('Webhook running on http://localhost:9001')
HTTPServer(('', 9001), H).serve_forever()
"
